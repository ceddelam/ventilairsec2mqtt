# DRAFT AppArmor profile for the ventilairsec2mqtt Home Assistant app. NOT ENABLED: it lives
# outside the app folder until it has been tested on a real Supervisor (a custom profile adds 1
# to the app's security rating, 7 → 8; a wrong one stops the app from working).
#
# Based on the profile of the Home Assistant apps security guide
# (https://contributors.home-assistant.io/apps/guides/security/). The Supervisor replaces the
# name on the top-level `profile` line with the app's ID (e.g. `local_ventilairsec2mqtt`).
#
# To test it in complain mode (allowed, but logged):
# 1. Copy it to the app folder as `apparmor.txt` (for a local app: into
#    `.local-app/ventilairsec2mqtt/` after `npm run app:local`), and add `complain` to the flags of
#    both profiles: flags=(attach_disconnected,mediate_deleted,complain).
# 2. Install or rebuild the app, run it for a while: stick connection, MQTT, an order, pairing,
#    a restart, a stop, a stick on `socket://` and on a serial device.
# 3. On the host, list what would have been denied:
#      journalctl _TRANSPORT="audit" -g 'apparmor="ALLOWED"'
#    (or `dmesg | grep apparmor`), add the missing rules, repeat.
# 4. Remove `complain`, check the app works in enforce mode, then move the file into the app
#    folder for good.

#include <tunables/global>

profile ventilairsec2mqtt flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities
  file,
  signal (send) set=(kill,term,int,hup,cont),

  # S6 overlay
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /etc/s6-overlay/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,

  # Bashio (scripts of the base image; the program itself does not use it)
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Options and state of the app
  /data/** rw,

  # The program: Node.js started by with-contenv, in its own, stricter profile
  /usr/bin/node cx -> ventilairsec2mqtt_node,

  profile ventilairsec2mqtt_node flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>
    #include <abstractions/nameservice>
    #include <abstractions/ssl_certs>

    # Stop request from S6 overlay
    signal (receive) peer=*_ventilairsec2mqtt,

    # MQTT (TCP), the Supervisor API (HTTP), a stick shared with ser2net (TCP), DNS
    network inet stream,
    network inet6 stream,
    network inet dgram,
    network inet6 dgram,
    network unix stream,
    network netlink raw,

    # Node.js and its libraries
    /usr/bin/node mrix,
    /usr/lib/** mr,
    /usr/share/icu/** r,
    /usr/share/zoneinfo/** r,
    /etc/localtime r,

    # The program
    /app/ r,
    /app/** r,
    /app/node_modules/@serialport/bindings-cpp/prebuilds/**.node mr,

    # Options (read) and state file (written through a temporary file renamed over it)
    /data/ r,
    /data/options.json r,
    /data/state.json rw,
    /data/state.json.* rw,

    # EnOcean stick: USB serial adapters and their stable names, locked by serialport
    /dev/ttyUSB[0-9]* rwk,
    /dev/ttyACM[0-9]* rwk,
    /dev/serial/ r,
    /dev/serial/** r,

    # What Node.js reads about its process and the system
    @{PROC}/@{pid}/** r,
    @{PROC}/sys/vm/overcommit_memory r,
    @{PROC}/meminfo r,
    @{PROC}/cpuinfo r,
    @{PROC}/stat r,
    /sys/fs/cgroup/** r,
    /sys/devices/system/cpu/** r,
  }
}
